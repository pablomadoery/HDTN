262  void compare(std::string const& dir, dir_entry_map& old_entries, dir_entry_map& new_entries)
263  {
264  // @todo filename() loses path relative to dir
265  // Need to construct with dir as a root_path() and get all paths relative to that.
266  for (dir_entry_map::iterator itn = new_entries.begin(); itn != new_entries.end(); ++itn)
267  {
268  dir_entry_map::iterator ito = old_entries.find(itn->first);
269  if (ito != old_entries.end())
270  {
271  bool pushBackEvent = false;
272  bool filesystemError = false;
273  for (unsigned int attempt = 0; attempt < 1000; ++attempt) {
274  try {
275  pushBackEvent = (!boost::filesystem::equivalent(itn->second.path(), ito->second.path()) ||
276  boost::filesystem::last_write_time(itn->second.path()) != boost::filesystem::last_write_time(ito->second.path()) ||
277  (boost::filesystem::is_regular_file(itn->second.path()) && boost::filesystem::is_regular_file(ito->second.path()) &&
278  boost::filesystem::file_size(itn->second.path()) != boost::filesystem::file_size(ito->second.path())));
279  }
280  catch (const boost::filesystem::filesystem_error&) {
281  filesystemError = true;
282  boost::this_thread::yield();
283  continue;
284  }
285  filesystemError = false;
286  break;
287  }
288  if (filesystemError) {
289  printf("boost::asio::dir_monitor_impl::compare: boost::filesystem::filesystem_error occurred after 1000 attempts calling boost::filesystem::equivalent\n");
290  }
291  if(pushBackEvent) {
292  pushback_event(dir_monitor_event(boost::filesystem::path(dir) / itn->second.path().filename(), dir_monitor_event::modified));
293  }
294  old_entries.erase(ito);
295  }
296  else
297  {
298  pushback_event(dir_monitor_event(boost::filesystem::path(dir) / itn->second.path().filename(), dir_monitor_event::added));
299  }
300  }
301  for (dir_entry_map::iterator it = old_entries.begin(); it != old_entries.end(); ++it)
302  {
303  pushback_event(dir_monitor_event(boost::filesystem::path(dir) / it->second.path().filename(), dir_monitor_event::removed));
304  }
305  }

307  void work_thread()
308  {
309 #ifndef EVFILT_USER
310  struct timespec timeout;
311  timeout.tv_sec = 0;
312  timeout.tv_nsec = 250000000; //250ms timeout
313 #endif
314 
315  static constexpr unsigned int MAX_EVENTS_OUT = 10;
316  struct kevent keventEventListOut[MAX_EVENTS_OUT];
317  while (true)
318  {
319 #ifndef EVFILT_USER
320  {
321  boost::unique_lock<boost::mutex> lock(events_mutex_);
322  if (!run_) {
323  break;
324  }
325  }
326 #endif
327  int nEvents = kevent(kqueue_, NULL, 0, keventEventListOut, (int)MAX_EVENTS_OUT,
328 #ifndef EVFILT_USER
329  &timeout);
330 #else
331  NULL); //blocks forever until event
332 #endif
333 
334  if (nEvents < 0)
335  {
336  printf("boost::asio::dir_monitor_impl::work_thread: kevent failed with nEvents == -1");
337  return;
338  }
339  for (int i = 0; i < nEvents; ++i) {
340  struct kevent& eventDataOut = keventEventListOut[i];
341  if (eventDataOut.flags == EV_ERROR) {
342  printf("boost::asio::dir_monitor_impl::work_thread: kevent failed with flags set to EV_ERROR");
343  return;
344  }
345 #ifdef EVFILT_USER
346  else if (eventDataOut.filter == EVFILT_USER) {
347  if (eventDataOut.ident == 0) {
348  //user event (id 0) triggered from destructor, time to exit work thread
349  return;
350  }
351  }
352 #endif
353  else if (eventDataOut.fflags & NOTE_WRITE) { //directory was modified
354  boost::unique_lock<boost::mutex> lock(dirs_mutex_);
355  dir_to_handle_bimap_t::right_iterator it = dirs_.right.find(eventDataOut.ident);
356  if (it != dirs_.right.end()) {
357  const std::string& directoryNameThatWasModified = it->second;
358  dir_entry_map new_entries;
359  scan(directoryNameThatWasModified, new_entries);
360  {
361  boost::unique_lock<boost::mutex> lock(entries_mutex_);
362  compare(directoryNameThatWasModified, entries_[directoryNameThatWasModified], new_entries);
363  std::swap(entries_[directoryNameThatWasModified], new_entries);
364  }
365  }
366  else {
367  printf("cannot find descriptor of directory that was modified\n");
368  }
369  }
370  else if (eventDataOut.fflags & (NOTE_DELETE | NOTE_RENAME)) {
371  printf("monitored directory was deleted or renamed\n");
372  remove_directory(eventDataOut.ident);
373  }
374  }
375  }
376  }
