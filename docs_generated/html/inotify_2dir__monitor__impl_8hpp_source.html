81  void check_sub_directory(const std::string &dirname, bool add_sub_directory)
82  {
83  boost::filesystem::directory_iterator end;
84  for (boost::filesystem::directory_iterator iter(dirname); iter != end; ++iter) {
85  if (boost::filesystem::is_directory(*iter)) {
86  if (add_sub_directory) {
87  try {
88  add_directory((*iter).path().string());
89  } catch (const std::exception&) {
90  continue;
91  }
92  } else {

165 private:
166  void end_read(const boost::system::error_code &ec, std::size_t bytes_transferred)
167  {
168  if (!ec)
169  {
170  pending_read_buffer_ += std::string(read_buffer_.data(), bytes_transferred);
171  while (pending_read_buffer_.size() > sizeof(inotify_event))
172  {
173  const inotify_event *iev = reinterpret_cast<const inotify_event*>(pending_read_buffer_.data());
174  dir_monitor_event::event_type type = dir_monitor_event::null;
175  switch (iev->mask)
176  {
177  case IN_CREATE: type = dir_monitor_event::added; break;
178  case IN_DELETE: type = dir_monitor_event::removed; break;
179  case IN_MODIFY: type = dir_monitor_event::modified; break;
180  case IN_MOVED_FROM: type = dir_monitor_event::renamed_old_name; break;
181  case IN_MOVED_TO: type = dir_monitor_event::renamed_new_name; break;
182  case IN_CREATE | IN_ISDIR:
183  {
184  type = dir_monitor_event::added;
185  add_directory(get_dirname(iev->wd) + "/" + iev->name);
186  break;
187  }
188  }
189  pushback_event(dir_monitor_event(boost::filesystem::path(get_dirname(iev->wd)) / iev->name, type));
190  pending_read_buffer_.erase(0, sizeof(inotify_event) + iev->len);
191  }
192 
193  begin_read();
194  }
195  else if (ec != boost::asio::error::operation_aborted)
196  {
197  boost::system::system_error e(ec);
198  boost::throw_exception(e);
199  }
200  }
