117  {
118  // The async_monitor thread will finish when async_monitor_work_ is reset as all asynchronous
119  // operations have been aborted and were discarded before (in destroy).
120  async_monitor_work_.reset();
121

123  async_monitor_io_service_.stop();

126  // destroyed _after_ the thread is finished (not that the thread tries to access
127  // instance properties which don't exist anymore).
128  async_monitor_thread_.join();
129  }
