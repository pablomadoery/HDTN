269  void work()
270  {
271  DWORD bytes_transferred = 0;
272  completion_key* ck = nullptr;
273  OVERLAPPED *overlapped = nullptr;
274 
275  helper::throw_system_error_if(!GetQueuedCompletionStatus(iocp_, &bytes_transferred, reinterpret_cast<PULONG_PTR>(&ck), &overlapped, INFINITE),
276  "boost::asio::basic_dir_monitor_service::work_thread: GetQueuedCompletionStatus failed");
277 
278  // a smart pointer is used to free allocated memory automatically in case of
279  // exceptions while handing over a completion key to the I/O completion port module,
280  // the ownership has to be *released* at the end of scope so as not to free the memory
281  // the OS kernel is using.
282  std::unique_ptr<completion_key> ck_holder(ck);
283  if (!ck_holder || !bytes_transferred)
284  return;
285 
286  // If a file handle is closed GetQueuedCompletionStatus() returns and bytes_transferred will be set to 0.
287  // The completion key must be deleted then as it won't be used anymore.
288 
289  // We must check if the implementation still exists. If the I/O object is destroyed while a directory event
290  // is detected we have a race condition. Using a weak_ptr and a lock we make sure that we either grab a
291  // shared_ptr first or - if the implementation has already been destroyed - don't do anything at all.
292  implementation_type impl = ck_holder->impl.lock();
293 
294  // If the implementation doesn't exist anymore we must delete the completion key as it won't be used anymore.
295  if (impl)
296  {
297  DWORD offset = 0;
298  PFILE_NOTIFY_INFORMATION fni = nullptr;
299  do
300  {
301  fni = reinterpret_cast<PFILE_NOTIFY_INFORMATION>(ck_holder->buffer + offset);
302  dir_monitor_event::event_type type = dir_monitor_event::null;
303  switch (fni->Action)
304  {
305  case FILE_ACTION_ADDED: type = dir_monitor_event::added; break;
306  case FILE_ACTION_REMOVED: type = dir_monitor_event::removed; break;
307  case FILE_ACTION_MODIFIED: type = dir_monitor_event::modified; break;
308  case FILE_ACTION_RENAMED_OLD_NAME: type = dir_monitor_event::renamed_old_name; break;
309  case FILE_ACTION_RENAMED_NEW_NAME: type = dir_monitor_event::renamed_new_name; break;
310  }
311  impl->pushback_event(dir_monitor_event(boost::filesystem::path(ck_holder->dirname) / std::wstring(fni->FileName, fni->FileNameLength / sizeof(WCHAR)), type));
312  offset += fni->NextEntryOffset;
313  } while (fni->NextEntryOffset);
314 
315  ZeroMemory(&ck_holder->overlapped, sizeof(ck_holder->overlapped));
316  helper::throw_system_error_if(!ReadDirectoryChangesW(ck_holder->handle,ck_holder->buffer, sizeof(ck_holder->buffer), FALSE, 0x1FF, &bytes_transferred, &ck_holder->overlapped, NULL),
317  "boost::asio::basic_dir_monitor_service::work_thread: ReadDirectoryChangesW failed");
318  }
319 
320  // if we come all along here, surviving all possible exceptions,
321  // the allocated memory has been successfully handed over to the OS,
322  // and we should let go of the ownership.
323  ck_holder.release();
324  }
