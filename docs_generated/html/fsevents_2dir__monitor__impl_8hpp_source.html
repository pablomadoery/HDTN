185  static void fsevents_callback(
186  ConstFSEventStreamRef streamRef,
187  void *clientCallBackInfo,
188  size_t numEvents,
189  void *eventPaths,
190  const FSEventStreamEventFlags eventFlags[],
191  const FSEventStreamEventId eventIds[])
192  {
193  (void)streamRef;
194  (void)eventIds;
195  char **paths = (char**)eventPaths;
196  dir_monitor_impl* impl = (dir_monitor_impl*)clientCallBackInfo;
197 
198  for (size_t i = 0; i < numEvents; ++i)
199  {
200  boost::filesystem::path dir(paths[i]);
201  //dir is an absolute file path with all symbolic links resolved
202  if (eventFlags[i] & kFSEventStreamEventFlagMustScanSubDirs) {
203  impl->pushback_event(dir_monitor_event(dir, dir_monitor_event::recursive_rescan));
204  }
205  if (eventFlags[i] & kFSEventStreamEventFlagItemCreated) {
206  if (!impl->already_added_directory(dir)) { //fix to ignore when creating directory right before adding to the monitor
207  impl->pushback_event(dir_monitor_event(dir, dir_monitor_event::added));
208  }
209  }
210  if (eventFlags[i] & kFSEventStreamEventFlagItemRemoved) {
211  impl->pushback_event(dir_monitor_event(dir, dir_monitor_event::removed));
212  }
213  if (eventFlags[i] & kFSEventStreamEventFlagItemModified) {
214  impl->pushback_event(dir_monitor_event(dir, dir_monitor_event::modified));
215  }
216  if (eventFlags[i] & kFSEventStreamEventFlagItemRenamed)
217  {
218  if (!boost::filesystem::exists(dir))
219  {
220  impl->pushback_event(dir_monitor_event(dir, dir_monitor_event::renamed_old_name));
221  }
222  else
223  {
224  impl->pushback_event(dir_monitor_event(dir, dir_monitor_event::renamed_new_name));
225  }
226  }
227  }
228  }
